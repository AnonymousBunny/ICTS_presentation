
\section{Proof sketch}
\begin{frame}
\textbf{Goal: }Upper bound $\text{spar}(g)$ in terms of $\text{rank}^{1/3} (g \circ \text{AND}_2)$

\begin{itemize}
	\item<1-> Assume, for the sake of contradiction, $\text{spar}(g)$ is large and $\text{rank}^{1/3} (g \circ \text{AND}_2)$ is small.
	\item<2-> From the second assumption, we have functions
	$$ a_i, b_i : \{0,1\}^n \rightarrow [0,1] , i= 1, 2, \cdots , m$$
	and constants $\alpha_i, i = 1, 2, \cdots , m$ such that
	
	$$\left| g(x \land y) - \displaystyle \sum_{i=1}^{m} \alpha_i a_i(x)  b_i(y)\right| \leq 1/3$$
	\item<2-> By standard techniques (eg John's lemma), we can assume $\sum |\alpha_i| \leq L$ (where $L \leq m^{O(n)}$)
\end{itemize}

\end{frame}

\begin{frame}
	\textbf{Assumption: }$\text{rank}^{1/3} (g \circ \text{AND}_2)$ is small, $\text{spar}(g)$ is large
	
	\begin{itemize}
		\item<1-> We shall construct a restriction $\rho$ such that:
		\begin{itemize}
			\item Approximate degree of $g_{\rho}$ is small
			\item Degree of $g_{\rho}$ is high
		\end{itemize}
		This contradicts known relations between degree and approximate degree (first proven by Nisan-Szegedy; constants were optimized later by Aaronson, Ben-David, Kothari, Rao, Tal). To be more precise, since $g_{\rho}$ is a Boolean function, $\tilde{deg}(g_{\rho}) \geq \Omega(\text{deg}(g_{\rho})^{1/2})$. We shall show that $g_{\rho}$ has much lower approximate degree than exact degree; therefore our assumption was impossible.
		
		\item<2-> We shall get a low-degree approximating polynomial from the inequality 
			$$\left| g(x \land y) - \displaystyle \sum_{i=1}^{m} \alpha_i a_i(x)b_i(y)\right| \leq 1/3$$
			whereas the high sparsity of $g$ will guarantee the original polynomial has high degree.
	\end{itemize}
	
\end{frame}


\begin{frame}
	\textbf{Assumption: }$\text{rank}^{1/3} (g \circ \text{AND}_2)$ is small, $\text{spar}(g)$ is large
	
	\begin{itemize}
		\item<1-> Note that every function $f: \{0,1\}^n \rightarrow \mathbb{R}$ can also be represented as a function $\tilde{f}: \{ \pm 1\}^n \rightarrow \mathbb{R}$. It is easy to see that their degrees are the same. \newline
		
		 \textbf{Convention: }For a function $f: \{0,1\}^n \rightarrow \mathbb{R}$, we denote its equivalent function with domain $\{ \pm 1\}^n$ by $\tilde{f}: \{ \pm 1\}^n \rightarrow \mathbb{R}$.
		\item<2-> For upper-bounding the approximate degree, we shall look at functions in the $\{\pm 1\}^n$ domain. For lower bounding the degree, we shall look at functions in the $\{ 0, 1\}^n$ domain. Since the degree is invariant under change of domain, this is ok.
	\end{itemize}
	
\end{frame}


\begin{frame}{Restriction+averaging operators}
	

	
	A \textbf{restriction+averaging operator} is denoted by a string $R \in \{0,1,*,\Delta\}^n$. It converts a function $f: \{0,1\}^n \rightarrow \mathbb{R}$ to a function $f: \{0,1\}^{\text{free}(R)} \rightarrow \mathbb{R}$ where $\text{free}(R)= \{i | R_i=*\}$. The transformation is as follows:
	\begin{itemize}
		\item If $R_i=b \in \{0,1\}$, fix the $i$-th coordinate to $b$.
		\item If $R_i= \Delta$, average over the $i$-th coordinate.
		\item If $R_i=*$, keep the $i$-th coordinate free.
	\end{itemize}
	
	A partial assignment $\rho \in \{0,1,*\}^n$ is said to be consistent with $R$ if it can be obtained from $R$ in the following manner:
	\begin{itemize}
		\item Change some of the 1's to 0's.
		\item Change all of the $\Delta$'s to 0.
	\end{itemize}
	
\end{frame}
\begin{frame}
	
	\begin{itemize}
		\item \textbf{Observation (will explain in more detail soon): }
		Let $R$ be a restriction+averaging operator and $\rho$ a consistent partial assignment. \newline
		In the expression 
		$$ g(x \land y) \approx \displaystyle \sum_{i=1}^{m} \alpha_i a_i(x)b_i(y),$$
		we can hit each $a_i$ by a restriction+averaging operator $R$ and replace each $b_i$ by a constant $c_i \in [0,1]$, and get an approximation for $g_{\rho}$:
		
		$$ g_{\rho}(z) \approx \displaystyle \sum_{i=1}^{m} \alpha_i c_i \times (a_i)_R(z)$$
	\end{itemize}
\end{frame}


\begin{frame}
	\textbf{Assumption: }$\text{rank}^{1/3} (g \circ \text{AND}_2)$ is small, $\text{spar}(g)$ is large. \newline
	\textbf{Goal: }Derive a contradiction
	\begin{itemize}
		\item<1-> We shall define a distribution $\mathcal{D}$ over $(R,\rho)$ ($\rho$ consistent with $R$) such that for any $\tilde{a}: \{ \pm 1\}^n \rightarrow [0,1]$,
		$$ \text{E}_{(R,\rho) \sim \mathcal{D}} [ |\tilde{a}_{R}|^{\geq d}] \leq \varepsilon$$
		
		where $|\tilde{f}|^{\geq d}$ denotes $\ell^1$ Fourier weight above level $d$
		\item<2-> ... and such that $\text{E}_{(R,\rho) \sim \mathcal{D}}[\text{deg}( g_{\rho})] \geq D$
	\end{itemize}	
	
\end{frame}

\begin{frame}
	\textbf{Assumption: }$\text{rank}^{1/3} (g \circ \text{AND}_2)$ is small, $\text{spar}(g)$ is large. \newline
\textbf{Goal: }Derive a contradiction
	\begin{itemize}
		\item<1-> The first property, along with the expression for approximate rank, will give that $\text{approx-deg}(g_{\rho}) \leq d$ for a typical $(R, \rho) \sim \mathcal{D}$
		\item<2-> The second property will give that $\text{deg}(g_{\rho}) \geq D$ for a typical $(R, \rho) \sim \mathcal{D}$.
		\item<3-> If the gap between $d$ and $D$ is large enough, we get a contradiction.
	\end{itemize}	
	
\end{frame}


\begin{frame}{Obtaining an approximation for $g_{\rho}$}
\begin{itemize}
	\item<1-> Recall the expression we got from the low-approximate rank assumption:
	$$ g(x \land y) \approx \displaystyle \sum_{i=1}^{m} \alpha_i a_i(x) b_i(y)$$
	$$ a_i, b_i: \{0,1\}^n \rightarrow [0,1]$$
	$$ \displaystyle \sum_i |\alpha_i| \leq L$$
	\item<2-> \textbf{Claim: }Given a restriction+averaging operator $R$ and a consistent partial assignment $\rho$, there exist constants $c_1, c_2, \cdots , c_m \in [0,1]$ such that
	$$ g_{\rho} (z) \approx \displaystyle \sum_{i=1}^{m} \alpha_i c_i \times (a_i)_{R} (z)$$

\end{itemize}	

\end{frame}

\begin{frame}{Obtaining an approximation for $g_{\rho}$}
	\begin{itemize}
		\item<1-> \textbf{Claim: }Given a restriction+averaging operator $R$ and a consistent partial assignment $\rho$ there exist constants $c_1, c_2, \cdots , c_m \in [0,1]$ such that
		$$ g_{\rho} (z) \approx \displaystyle \sum_{i=1}^{m} \alpha_i c_i \times (a_i)_{R} (z)$$
		\item<2-> I will show this only for two special cases. The general case should be clear from this.

	\end{itemize}	
	
\end{frame}


\begin{frame}{Obtaining an approximation for $g_{\rho}$}
	\begin{itemize}
		\item<1->\textbf{Special case 1: } $R= \{\Delta , *, *, *, \cdots , *\}, \rho = \{0, *, *, *, \cdots , *\}$.
		\item<2-> Plugging $y_0= (0, 1, 1, \cdots , 1), x= (1, z_2, z_3, \cdots , z_n)$, we get
		$$ g(0, z_2, z_3, \cdots , z_n) \approx \displaystyle \sum_{i=1}^{m} \alpha_i a_i(1, z_2, \cdots , z_n) b_i(y_0)$$
		Plugging $y_0= (0, 1, 1, \cdots , 1), x= (0, z_2, z_3, \cdots , z_n)$,
		$$ g(0, z_2, z_3, \cdots , z_n) \approx \displaystyle \sum_{i=1}^{m} \alpha_i a_i(0, z_2, z_3, \cdots , z_n) b_i(y_0) $$
		\item<3-> Average these two expressions:
		$$ g(0, z_2, \cdots , z_n) \approx \displaystyle \sum_{i=1}^{m} \alpha_i b_i(y_0) \times (a_i)_R(z)$$
			
	\end{itemize}	
	
\end{frame}

\begin{frame}{Obtaining an approximation for $g_{\rho}$}
	\begin{itemize}
		\item<1->\textbf{Special case 2: } $R= \{1 , *, *, *, \cdots , *\}, \rho = \{0, *, *, *, \cdots , *\}$.
		\item<2-> Plugging $y_0= (0, 1, 1, \cdots , 1), x= (1, z_2, z_3, \cdots , z_n)$, we get
		$$ g(0, z_2, z_3, \cdots , z_n) \approx \displaystyle \sum_{i=1}^{m} \alpha_i a_i(1, z_2, \cdots , z_n) b_i(y_0)$$
		This is the desired expression.
		
	\end{itemize}	
	
\end{frame}



\begin{frame}{Exhibiting an impossible gap between degree and approximate degree}
	\begin{itemize}
		\item<1-> \textbf{Conclusion: }If $R$ is a restriction+averaging operator and $\rho$ is a consistent partial assignment, we get an approximation for $g_{\rho}$ in terms of the $(a_i)_R$'s.
		\item<2-> \textbf{Our next goal: }Using the fact that sparsity of $g$ is high, we define a distribution $\mathcal{D}$ over $(R,\rho)$ such that:
		\begin{itemize}
			\item $\rho$ is consistent with $R$
			\item For all $a: \{0,1\}^n \rightarrow [0,1]$, $\text{E}_{R, \rho \sim \mathcal{D}}[\text{approx-deg}(a_{R})] \leq d$ \textcolor{red}{(we will do something slightly different actually)}
			\item $\text{E}_{(R, \rho )\sim \mathcal{D}} [\text{deg}(g_{\rho})] \geq D$ 
		\end{itemize} 
	\end{itemize}
\end{frame}
\begin{frame}{Exhibiting an impossible gap between degree and approximate degree}
	\begin{itemize}		
		
		\item<1-> If we can do this with a sufficiently big gap between $d$ and $D$, from the previous observation we get that the approximate degree of $g_{\rho}$ is very small; whereas the second property implies the exact degree of $g_{\rho}$ is very large. But Nisan-Szegedy\footnote{\textcolor{red}{TODO: explain on the board in case Yogesh has not covered this}} implies for any Boolean function, degree and approximate degree are polynomially related . Contradiction.
		
		
	\end{itemize}	
	
\end{frame}



\begin{frame}{Simplifying any bounded function}
\textbf{First requirement: }
\begin{itemize}
\item<1-> This is what we will show: \newline
For any $\tilde{a}: \{ \pm 1\}^n \rightarrow [0,1]$, 
$$ \text{E}_{R, \rho \sim \mathcal{D}} [|| \tilde{a}_R ||^{\geq d}] \leq \varepsilon$$
where $||\tilde{f}||^{\geq d}$ denotes total $\ell^1$ Fourier mass above level $d$. \newline
Notice that we switched to $\{ \pm 1\}^n$ domain here.
\item<2-> If we achieve this, the expected $\ell^1$ Fourier mass above level $d$ of the function
$$ \displaystyle \sum_{i=1}^{L} \alpha_i c_i (a_i)_R$$ will be at most $L \varepsilon$, which we will set to be $<1/100$. Thus, on expectation, the approximate degree of $g_{\rho}$ will be $\leq d$.
\end{itemize}
\end{frame}

\begin{frame}{Simplifying any bounded function}
\textbf{Why switching to the $\{ \pm 1 \}^n$ basis is helpful:}
\begin{itemize}
	\item<1-> Let $\tilde{f}: \{ \pm 1\}^n \rightarrow \mathbb{R}$ be any function. Its Fourier expansion is
	$$ \tilde{f} = \displaystyle \sum_{S \subseteq [n]} \hat{\tilde{f}}(S) \chi_S$$
	\item<2-> Suppose we hit $\tilde{f}$ by an averaging operator $R$ which averages over some subset $T$ of variables. Then, we have
	$$ \tilde{f}_R = \displaystyle \sum_{S \cap T = \phi} \hat{\tilde{f}}(S) \chi_S$$
	\item<3->In words, averaging out over a variable kills all monomials containing it.
\end{itemize}
\end{frame}

\begin{frame}{Defining the distribution}

\begin{itemize}
	\item<1-> Assume $g: \{0,1\}^n \rightarrow \{0,1\}$ has $\geq s$ non-zero monomials in the de-Morgan basis. Let $\mathcal{F} \subseteq 2^{[n]}$ denote the set of nonzero monomials. 
	\item<2-> By \textbf{Sauer-Shelah lemma}, the VC dimension of $\mathcal{F}$ is at least $\Omega (\log(s)/\log(n))$.
	\item<3-> Let $T \subseteq [n]$ be a shattered set of size $\geq \Omega(\log(s)/\log(n))$.
	\item<4-> For any $A \subseteq T$, there exists $S \in \mathcal{F}$ such that $S \cap T = A$.
\end{itemize}

\end{frame}


\begin{frame}{Defining the distribution}
	
	\begin{itemize}
		\item<1-> Assume $g: \{0,1\}^n \rightarrow \{0,1\}$ has $\geq s$ non-zero monomials in the de-Morgan basis. Let $\mathcal{F} \subseteq 2^{[n]}$ denote the set of nonzero monomials. 
		\item<2-> By \textbf{Sauer-Shelah lemma}, the VC dimension of $\mathcal{F}$ is at least $\Omega (\log(s)/\log(n))$.
		\item<3-> Let $T \subseteq [n]$ be a shattered set of size $\geq \Omega(\log(s)/\log(n))$.
		\item<4-> For any $A \subseteq T$, there exists $S \in \mathcal{F}$ such that $S \cap T = A$.
	\end{itemize}
	
\end{frame}

\begin{frame}{Defining the distribution}
	
	\begin{itemize}
		\item<1-> Let $T$ be the large shattered subset.
		\item<2-> For every $i \in T$, set 
		$$
		R(i) = \begin{cases}
			* & \text{ with probability }p \\
			\Delta & \text{ with probability }1-p
		\end{cases}
		$$
		Set $R(i)=1$ for all $i \not \in T$.
		
	(Here $p$ is small but non-negligible.)
\end{itemize}\end{frame}
\begin{frame}{Defining the distribution}
	\begin{itemize}
	\item<1-> Let $A = \{i | i \in T, R(i)=*\}. $ Set 
	$$ \rho(i)= \begin{cases}
		* & \text{ for }i \in A \\
		0 & \text{for }i \in T \setminus A
	\end{cases}$$
	\item<2->
	Look at the polynomial
	$$ g_{A,T} = \displaystyle \sum_{S : S \cap T = A } \alpha_S x_{S \setminus A}$$
	This is defined on $\{0,1\}^{[n] \setminus T}$, and is nonzero since $T$ is shattered. So there exists a $y \in \{0,1\}^{[n] \setminus T}$ such that $g_{A,T}(y) \neq 0$. 
	\item<4-> Set $\rho(i)=y_i$ for $i \not \in T$.
	\end{itemize}
\end{frame}

\begin{frame}{this distribution is good}
	Recall that we have to show the following properties about this distribution:
	\begin{enumerate}
		\item<1-> \textbf{It simplifies any bounded function: }For any $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
		$$ \text{E} [|\hat{\tilde{f}}|^{\geq d}] \leq \varepsilon$$
		\item<2-> \textbf{It preserves high degree of $g$}:
		$$ \text{E}[\text{deg} (g_{\rho})] \geq D$$ 
	\end{enumerate}
	
\end{frame}


\begin{frame}{Preserving high degree}
	\textbf{Claim: }$\text{E} [\text{deg}(g_{\rho})] \geq p|T|$
	\begin{proof}
		The coefficient of $x_A$ in $g_{\rho}$ is nonzero, and we have $\text{E}[|A|] = p|T|$. \newline
		
		The coefficient of $x_A$ in $g_{\rho}$ is $g_{A,T}(y)= \displaystyle \sum_{S\cap T=A} \alpha_S y_{S \setminus A}$; and we chose $y$ so that it is nonzero.
	\end{proof}
	
\end{frame}


\begin{frame}{Destroying all bounded functions}
	\textbf{Claim: }For all $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
	$$ \text{E} [|| \hat{\tilde{f}}_R ||^{\geq d}] \leq \varepsilon$$
	\begin{proof}
		Recall how $R$ is sampled:
		\begin{itemize}
			\item Set everything outside $T$ to 1.
			\item Set everything in $T$ independently to $*$ with probability $p$, $\Delta$ with probability $1-p$.
		\end{itemize}
	\end{proof}
	
\end{frame}

\begin{frame}{Destroying all bounded functions}
	\textbf{Claim: }For all $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
	$$ \text{E} [|| \hat{\tilde{f}}_R ||^{\geq d}] \leq \varepsilon$$
	\begin{proof}
		\begin{itemize}
		\item<1-> So we can first think of hitting $f$ with the fixed restriction $\delta$ that fixes everything outside $T$ to 1. Let $h= f_{\delta}$. We set every variable of $h$ to $*$ with probability $p$; otherwise we set it to $\Delta$.
		\item<2-> If a monomial of $\tilde{h}$ contains any variable that gets hit by $\Delta$, that monomial dies.
		\item<3-> The probability of a monomial $\hat{\tilde{h}}(S) \chi_S$ surviving, therefore, is $p^{|S|}$. For large monomials this is small.
		\end{itemize}
	\end{proof}
	
\end{frame}


\begin{frame}{Destroying all bounded functions}
	\textbf{Claim: }For all $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
	$$ \text{E} [|| \hat{\tilde{f}}_R ||^{\geq d}] \leq \varepsilon$$
	\begin{proof}
		\begin{itemize}
			\item<1-> 
			$$ \text{Expected Fourier mass above level }d \leq \displaystyle \sum_{|S| \geq d} p^{|S|} |\hat{\tilde{h}}(S)|$$
			\item<2-> Break this sum by layers $k= d, d+1, \cdots , n$.  At layer $k$, the sum is
			$$ p^{k} \displaystyle \sum_{|S|=k} |\hat{\tilde{h}}(S)|$$			
		\end{itemize}
	\end{proof}
	
\end{frame}


\begin{frame}{Destroying all bounded functions}
	\textbf{Claim: }For all $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
	$$ \text{E} [|| \hat{\tilde{f}}_R ||^{\geq d}] \leq \varepsilon$$
	\begin{proof}
		\begin{itemize}
			\item<1-> 
			 At layer $k$, the sum is
			$$ p^{k} \displaystyle \sum_{|S|=k} |\hat{\tilde{h}}(S)|$$		
			\item<2-> Since $\sum_S \hat{\tilde{h}}(S)^2 \leq 1$, by Cauchy-Schwartz, the sum above is 
			$$ \leq p^{k} \sqrt{\dbinom{n}{k}} \leq \left(p \sqrt{\dfrac{en}{k}}  \right)^k$$	
		\end{itemize}
	\end{proof}
	
\end{frame}

\begin{frame}{Destroying all bounded functions}
	\textbf{Claim: }For all $\tilde{f}: \{ \pm 1\}^n \rightarrow [0,1]$,
	$$ \text{E} [|| \hat{\tilde{f}}_R ||^{\geq d}] \leq \varepsilon$$
	\begin{proof}
		\begin{itemize}
			\item<1-> 
			Note that $k \geq d$. Choose $d \approx n^{1/100}$, $p \approx d^{-99/200 - \epsilon}$. Win.
		\end{itemize}
	\end{proof}
	
\end{frame}
